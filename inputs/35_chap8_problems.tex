\subsection*{Problems}

\textbf{8.1.} Let $\mathcal{X}$ and $\mathcal{Y}$ be the set of natural numbers $\mathbb{N}$ and the $\sigma$-algebras the power set $\mathcal{P}(\mathbb{N})$. Let $\mu$ and $\nu$ be the counting measures on $\mathcal{X}$ and $\mathcal{Y}$, respectively. We define a function $f:\mathcal{X}\times \mathcal{Y}\to \mathbb{R}$ by
\[
f(i,j)\triangleq
\begin{cases}
i & \text{if } i=j,\\
-i & \text{if } j=i+1,\\
0 & \text{otherwise.}
\end{cases}
\]

Show that the conclusion of the Fubini theorem fails in this example.

\textbf{8.2.} Apply Fubini's theorem to prove that the order of summation in $\sum_{i=0}^{\infty}\sum_{j=0}^{\infty} a_{ij}$ is irrelevant if $\sum_{i=0}^{\infty}\sum_{j=0}^{\infty} |a_{ij}|$ is finite.

\textbf{8.3.} Let $P$ and $Q$ be exponential distributions with parameters $\lambda$ and $\mu$, respectively. Find the total variation distance between $P$ and $Q$.

\textbf{8.4.} Define two probability distributions $P$ and $Q$ on a finite sample space $\Omega=\{a,b,c\}$. The pmf's of $P$ and $Q$ are given in the following table:
\[
\begin{array}{c|ccc}
\omega & a & b & c\\
\hline
P(\omega) & 1/3 & 1/3 & 1/3\\
Q(\omega) & 1/2 & 1/4 & 1/4
\end{array}
\]

Find the maximal coupling of $P$ and $Q$.

\textbf{8.5.} Let $\mu$ and $\nu$ be continuous-type probability measures with pdf $f(x)$ and $g(x)$, respectively. Show that for any bounded measurable function $h$ and Borel set $B$,
\[
\left|
\int_B h(x)f(x)\, dx - \int_B h(x)g(x)\, dx
\right|
\le
2d_{TV}(\mu,\nu)\cdot \sup_{x\in B} h(x).
\]

\textbf{8.6 (Poisson Approximation of Binomial Distribution).} Suppose $X=X_1+X_2+\cdots+X_n$ and $Y=Y_1+Y_2+\cdots+Y_n$ are sums of random variables. We use the notation $\operatorname{Ber}(\lambda)$ for Bernoulli distribution with mean $\lambda$ and $\operatorname{Poi}(\lambda)$ for Poisson distribution with mean $\lambda$.
\begin{enumerate}[label=(\alph*)]
    \item Show that $P(X\neq Y)\le \sum_{k=1}^{n} P(X_k\neq Y_k)$. (Show that the event $\{X\neq Y\}$ is a subset of $\cup_{k=1}^{n}\{X_k\neq Y_k\}$.)
    \item Construct a maximal coupling of $\operatorname{Ber}(\lambda)$ and $\operatorname{Poi}(\lambda)$.
    \item For $k=1,2,\dots,n$, let $\hat{X}_k$ and $\hat{Y}_k$ be the maximal coupling of $\operatorname{Ber}(\lambda_k)$ and $\operatorname{Poi}(\lambda_k)$ obtained in part (b). Show that the total variation between $\hat{X}=\hat{X}_1+\cdots+\hat{X}_n$ and $\hat{Y}=\hat{Y}_1+\cdots+\hat{Y}_n$ is upper bounded by $\sum_{k=1}^{n} \lambda_k^2$.
    \item Derive an upper bound of the total variation distance between $\operatorname{Binom}(n,\lambda/n)$ and $\operatorname{Poi}(\lambda)$ by substituting $\lambda_k=\lambda/n$ in part (c).
\end{enumerate}

\textbf{8.7.} Let $X$ and $Y$ be continuous-type random variables with pdf $f_X(x)$ and $f_Y(y)$, respectively. Suppose both $f_X(x)$ and $f_Y(y)$ are piece-wise continuous. We construct random variables $\hat{X}$ and $\hat{Y}$ as follows. First compute
\[
p \triangleq \frac{1}{2}\int_{-\infty}^{\infty} |f_X(t)-f_Y(t)|\, dt.
\]
Let $B$ be a Bernoulli random variable with success probability $p$. Define three more auxiliary random variables $U$, $V$, and $W$ with the following pdf's:
\[
f_U(u)=\min(f_X(u),f_Y(u))/(1-p)
\]
\[
f_V(v)=(f_X(v)-\min(f_X(v),f_Y(v)))/p
\]
\[
f_W(w)=(f_Y(w)-\min(f_X(w),f_Y(w)))/p.
\]

Generate random variables $B$, $U$, $V$, and $W$ such that they are mutually independent. Define $(\hat{X},\hat{Y}):=(U,U)$ if $B=0$ and $(\hat{X},\hat{Y}):=(V,W)$ if $B=1$. Prove that this gives a maximum coupling of the pdf $f_X$ and $f_Y$.
